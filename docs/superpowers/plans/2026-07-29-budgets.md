# Budgets (Per-Category, Pay-Cycle-Native) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a user set a spending limit per expense category, and see progress against it — full list on a dedicated Budgets screen, near-limit callouts as progress bars on Dashboard.

**Architecture:** No new SwiftData model — `CategoryModel.monthlyBudget: Decimal?` already exists in the schema and already flows through `CategorySnapshot`/`CategoryInput`/`TransactionActor`, it's just never been surfaced in UI. Budget progress is computed live (no persistence, no caching): `PayCycleService.currentFinancialMonth` gives the current cycle window, a new `BudgetProgressService` sums that window's expense transactions per category from the already-fetched `[TransactionSnapshot]`. The Budgets *editing* screen follows the same direct-SwiftData `@Query` + `modelContext.save()` pattern `CategorySettingsView`/`EditCategorySheet` already use (no repository plumbing needed there). The Dashboard *display* of near-limit budgets follows `DashboardViewModel`'s existing `computeAnomalyCallout`-style nonisolated static function pattern for testability.

**Tech Stack:** SwiftUI, SwiftData, Swift Testing (`@Test`, `#expect`), MVVM, repository pattern (`ITransactionRepository`).

## Global Constraints

- Per-category budgets only — no overall/whole-ledger budget (per brainstorm decision).
- Budgets screen reached from the Profile bottom sheet, not inline on Categories (per brainstorm decision).
- Dashboard shows progress bars only for **near-limit** categories (≥80% of budget spent); full list lives on the Budgets screen (per brainstorm decision).
- Categories with no budget set (`monthlyBudget == nil` or `0`) don't appear in any budget UI.
- Amounts stored as `Decimal`, expenses are negative `TransactionSnapshot.amount` — budget math must use `abs()` on expense amounts, never assume sign.
- EUR hardcoded throughout; no multi-currency handling needed here.
- Build/test only via Xcode MCP tools (`mcp__xcode__BuildProject`, `mcp__xcode__RunSomeTests`) with `tabIdentifier: "windowtab1"` — never `xcodebuild` in Bash. Load schemas first via `ToolSearch` with `query: "select:mcp__xcode__BuildProject,mcp__xcode__RunAllTests,mcp__xcode__RunSomeTests"`.

```swift
//  BudgetProgressService.swift
//  PersonalFinanceTraker

import Foundation
import SwiftData

struct BudgetProgress: Identifiable, Equatable, Sendable {
    let id: PersistentIdentifier
    let categoryName: String
    let systemImage: String
    let colorToken: String
    let budget: Decimal
    let spent: Decimal

    var percent: Double {
        guard budget > 0 else { return 0 }
        return Double(truncating: (spent / budget) as NSDecimalNumber)
    }

    var isOverBudget: Bool { spent > budget }
}

enum BudgetProgressService {
    static func computeProgress(
        categories: [CategorySnapshot],
        transactions: [TransactionSnapshot],
        payCycleStartDay: Int,
        calendar: Calendar = .current
    ) -> [BudgetProgress] {
        let budgeted = categories.filter {
            $0.transactionType == .expense && ($0.monthlyBudget ?? 0) > 0
        }
        guard !budgeted.isEmpty else { return [] }

        let (cycleStart, cycleEnd) = PayCycleService.currentFinancialMonth(
            startDay: payCycleStartDay, calendar: calendar
        )

        var spentByCategory: [String: Decimal] = [:]
        for tx in transactions where tx.amount < 0 {
            guard tx.timestamp >= cycleStart && tx.timestamp <= cycleEnd else { continue }
            spentByCategory[tx.category, default: 0] += abs(tx.amount)
        }

        return budgeted.map { category in
            BudgetProgress(
                id: category.persistentId,
                categoryName: category.name,
                systemImage: category.systemImage,
                colorToken: category.colorToken,
                budget: category.monthlyBudget ?? 0,
                spent: spentByCategory[category.name] ?? 0
            )
        }
    }

    static func nearLimit(_ progress: [BudgetProgress], threshold: Double = 0.8) -> [BudgetProgress] {
        progress.filter { $0.percent >= threshold }.sorted { $0.percent > $1.percent }
    }
}
```

import SwiftUI
import SwiftData

struct BudgetRow: View {
    @Bindable var category: CategoryModel
    @Environment(\.modelContext) private var modelContext
    @FocusState private var isFocused: Bool
    @State private var text: String = ""

    var body: some View {
        HStack(spacing: 12) {
            Image(systemName: category.systemImage)
                .foregroundStyle(category.categoryColor)
                .frame(width: 24)
            Text(category.name)
                .foregroundStyle(.textPrimary)
            Spacer()
            TextField("No limit", text: $text)
                .keyboardType(.decimalPad)
                .multilineTextAlignment(.trailing)
                .focused($isFocused)
                .frame(width: 100)
                .foregroundStyle(.textPrimary)
        }
        .onAppear { text = displayText(for: category.monthlyBudget) }
        .onChange(of: isFocused) { _, focused in
            if !focused { commit() }
        }
    }

    private func displayText(for budget: Decimal?) -> String {
        guard let budget, budget > 0 else { return "" }
        return NSDecimalNumber(decimal: budget).stringValue
    }

    private func commit() {
        let normalized = text.replacingOccurrences(of: ",", with: ".")
        if let value = Decimal(string: normalized), value > 0 {
            category.monthlyBudget = value
        } else {
            category.monthlyBudget = nil
            text = ""
        }
        try? modelContext.save()
    }
}
